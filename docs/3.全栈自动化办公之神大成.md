## 案例 1：从 Excel 读取数据填入系统

**场景**：从 Excel 文件中读取客户信息，逐条填入 CRM 系统。

**学习要点**：`pandas` 读取 Excel、`for_data` 循环填入、鲁棒点击。

```python
from imgclickflow import auto
import pandas as pd

# 读取 Excel 数据
df = pd.read_excel("新客户信息.xlsx")

auto.click("CRM系统")
auto.wait("客户管理界面")

for _, row in df.iterrows():
    auto.do()
        .click("新增客户")
        .write(str(row["客户名称"]))
        .press("tab")
        .write(str(row["联系电话"]))
        .press("tab")
        .write(str(row["联系地址"]))
        .click_robust("保存", retries=2)
        .wait("保存成功")
        .click("返回列表")
        .run()

auto.shot("录入完成")
```


## 案例 2：配置表驱动多场景切换

**场景**：一份配置 Excel 中定义了不同场景下需要依次点击的模块序列。用户选定场景后，脚本自动按序点击各模块并截图。

**学习要点**：`pandas` 读取配置、`click_seq` 序列点击、`snap` 流程内截图。

配置 Excel `场景配置.xlsx` 内容：

| 场景名称 | 模块序列 |
|:---|:---|
| 生产环境 | 登录, 生产看板, 质量检测, 设备状态 |
| 测试环境 | 登录, 测试看板, 缺陷管理, 自动化报告 |
| 演示环境 | 登录, 数据概览, 趋势分析 |

```python
from imgclickflow import auto
import pandas as pd

# 读取配置
config_df = pd.read_excel("场景配置.xlsx")
config_map = dict(zip(config_df["场景名称"], config_df["模块序列"]))

# 选择场景
scene = "生产环境"
module_list = config_map[scene].split(", ")

# 依次点击各个模块，每个模块截一张图
auto.do()
    .click_seq(module_list)          # 依次点击所有模块
    .snap(f"{scene}-巡检完成")        # 流程内截图留证
    .run()

auto.shot(f"{scene}-全屏截图")
```


## 案例 3：从 API 拉取数据并填入，自动邮件通知

**场景**：从公司 API 获取今日巡检任务清单，逐条填入巡检系统，完成后自动发送邮件给主管。

**学习要点**：`requests` 调 API、`smtplib` 发邮件、鲁棒点击。

```python
from imgclickflow import auto
import requests
import smtplib
from email.mime.text import MIMEText
import datetime

# 1. 从 API 获取数据
response = requests.get("https://api.example.com/patrol/tasks?date=today",
                        headers={"Authorization": "Bearer your_token"})
tasks = response.json()["tasks"]

# 2. 登录巡检系统
auto.click("巡检系统")
auto.wait("登录界面")
auto.write("admin")
auto.press("tab")
auto.write("password")
auto.press("enter")
auto.wait("主界面")

# 3. 逐条填入任务
for task in tasks:
    auto.do()
        .click("新增记录")
        .write(task["device_name"])
        .press("tab")
        .write(task["check_item"])
        .press("tab")
        .write("正常" if task["status"] == "ok" else "异常")
        .click_robust("保存记录", retries=2)
        .wait("保存成功")
        .click("继续")
        .run()

# 4. 截图存档
auto.shot(f"巡检记录_{datetime.date.today()}")

# 5. 发送邮件通知
msg = MIMEText(f"今日巡检已完成，共处理 {len(tasks)} 项任务。")
msg["Subject"] = f"巡检报告 - {datetime.date.today()}"
msg["From"] = "your@email.com"
msg["To"] = "manager@email.com"

with smtplib.SMTP_SSL("smtp.qq.com", 465) as server:
    server.login("your@email.com", "your_auth_code")
    server.send_message(msg)

auto.shot("流程全部完成")
```


## 案例 4：从数据库读取并生成报表

**场景**：连接公司数据库，查询上周销售数据，汇总后生成 Excel 报表，自动导入桌面报表软件预览打印。

**学习要点**：`pymysql` 查询数据库、`pandas` 数据处理、`openpyxl` 生成报表。

```python
from imgclickflow import auto
import pymysql
import pandas as pd
from datetime import date, timedelta

# 1. 从数据库查询数据
conn = pymysql.connect(host="192.168.1.100", user="readonly",
                       password="password", database="sales_db")
sql = """
    SELECT product_name, SUM(amount) as total_amount, COUNT(*) as order_count
    FROM sales_orders
    WHERE order_date BETWEEN %s AND %s
    GROUP BY product_name
    ORDER BY total_amount DESC
"""
end_date = date.today() - timedelta(days=1)
start_date = end_date - timedelta(days=7)
df = pd.read_sql(sql, conn, params=(start_date, end_date))
conn.close()

# 2. 在 Python 中生成报表
summary = df.head(10)
summary.to_excel("销售周报.xlsx", index=False)

# 3. 通过图像自动化打开报表软件并导入
auto.click("报表系统")
auto.click("数据导入")
auto.write("C:\\自动化脚本\\销售周报.xlsx")
auto.press("enter")
auto.click_robust("开始导入", retries=3)
auto.wait("导入完成")

# 4. 打印预览并截图
auto.click("打印预览")
auto.shot(f"销售周报_{start_date}_{end_date}")
```


## 案例 5：Web 下载 + 桌面处理（双引擎）

**场景**：从 Web 管理后台下载每日报表 Excel，另存后用桌面软件打开处理。

**学习要点**：Playwright 负责 Web 段操作，ImgClickFlow 负责桌面段操作，`glob` 查找最新下载文件。

```python
from imgclickflow import auto
from playwright.sync_api import sync_playwright
import glob
import os
import datetime
import time as time_module

# 步骤 1：Playwright 负责 Web 登录→下载
with sync_playwright() as p:
    browser = p.chromium.launch(headless=False)
    page = browser.new_page()
    page.goto("https://admin.example.com/login")
    page.fill("input[name='username']", "operator")
    page.fill("input[name='password']", "password123")
    page.click("button[type='submit']")
    page.wait_for_selector("text=管理后台")

    # 下载今日报表
    page.click("text=日报管理")
    page.click("text=下载", timeout=5000)
    with page.expect_download() as download_info:
        page.click("text=Excel格式")
    download = download_info.value
    download.save_as(f"下载/日报_{datetime.date.today()}.xlsx")
    browser.close()

# 步骤 2：ImgClickFlow 负责桌面操作
time_module.sleep(1)

# 查找最新下载的 Excel 文件
download_folder = os.path.expanduser("~\\Downloads")
files = glob.glob(os.path.join(download_folder, "日报_*.xlsx"))
latest_file = max(files, key=os.path.getctime)

auto.click("文件夹")
auto.write(download_folder)
auto.press("enter")
time_module.sleep(0.5)
auto.dclick(latest_file.split("\\")[-1][:35])     # 双击文件名打开
auto.wait("Excel")
auto.click_robust("打印预览")
auto.shot(f"日报_{datetime.date.today()}_预览")
```


## 案例 6：Word 文档批量生成

**场景**：根据 Excel 中的巡检数据，批量生成 Word 格式的巡检报告，每台设备一个文档。

**学习要点**：`python-docx` 创建 Word、`pandas` 读取数据、`for_data` 循环生成。

```python
from imgclickflow import auto
import pandas as pd
from docx import Document
import datetime

# 读取巡检数据
df = pd.read_excel("巡检记录.xlsx")

# 逐行生成 Word 报告
for _, row in df.iterrows():
    doc = Document()
    doc.add_heading(f"巡检报告 - {row['设备名称']}", level=1)
    doc.add_paragraph(f"巡检日期：{datetime.date.today()}")
    doc.add_paragraph(f"设备状态：{row['状态']}")
    doc.add_paragraph(f"检查项目：{row['检查项目']}")
    doc.add_paragraph(f"备注：{row['备注']}")
    doc.save(f"巡检报告_{row['设备名称']}.docx")

# 通过 ImgClickFlow 打开文件夹查看结果
auto.click("文件夹")
auto.write("C:\\自动化脚本\\")
auto.press("enter")
auto.shot("批量生成完成")
```


## 案例 7：PDF 表格提取并填入系统

**场景**：从供应商发来的 PDF 发票中提取表格数据，自动填入财务系统。

**学习要点**：`pdfplumber` 提取表格、`pandas` 清洗数据、鲁棒点击填入。

```python
from imgclickflow import auto
import pdfplumber
import pandas as pd

# 1. 从 PDF 提取表格
with pdfplumber.open("发票.pdf") as pdf:
    page = pdf.pages[0]
    tables = page.extract_tables()
    # 假设表格第一行为表头，后续为数据
    data = tables[0][1:]  # 跳过表头

# 2. 登录财务系统
auto.click("财务系统")
auto.wait("登录界面")
auto.write("finance")
auto.press("tab")
auto.write("password")
auto.press("enter")
auto.wait("主界面")

# 3. 逐行填入
auto.click("新增发票")
for row in data:
    invoice_no, product, amount, date = row[0], row[1], row[2], row[3]
    auto.do()
        .click("添加明细")
        .write(str(invoice_no))
        .press("tab")
        .write(str(product))
        .press("tab")
        .write(str(amount))
        .press("tab")
        .write(str(date))
        .click_robust("保存明细", retries=2)
        .wait("保存成功")
        .run()

auto.click("提交发票")
auto.click_robust("确认提交")
auto.wait("提交成功")
auto.shot("发票录入完成")
```
